\section{整除与因式}\label{sec:1.3}

\begin{ex}{1}\label{ex:1.3.1}
    \(f\p{x}=x^4+x^3+x^2+x+1,g\p{x}=x^2+2x+1\), 求 \(f\p{x}\) 除以 \(g\p{x}\) 的商和余式.
\end{ex}

\begin{solution}
    对 \(f\p{x}\) 除以 \(g\p{x}\) 进行多项式长除法:
    \[
    \begin{array}{r @{\;} l @{\;} c @{\;} c @{\;} c @{\;} c @{\;} c @{\;} c @{\;} c @{\;} c @{\;} c}
        & & x^2 & - & x & + & 2 & & & & \\
        \cline{2-11}
        x^2+2x+1 & \big) & x^4 & + & x^3 & + & x^2 & + & x & + & 1 \\
        & & x^4 & + & 2x^3 & + & x^2 & & & & \\
        \cline{3-11}
        & & & - & x^3 & & & + & x & + & 1 \\
        & & & - & x^3 & - & 2x^2 & - & x & & \\
        \cline{4-11}
        & & & & & & 2x^2 & + & 2x & + & 1 \\
        & & & & & & 2x^2 & + & 4x & + & 2 \\
        \cline{7-11}
        & & & & & & & - & 2x & - & 1
    \end{array}
    \]
    因此, 商为 \(x^2-x+2\), 余式为 \(-2x-1\).
\end{solution}

\begin{ex}{3}\label{ex:1.3.3}
    设 \(f,g,h\in\Omega\br{x}\), 证明 \(h\mid f-g\) 当且仅当 \(f\) 与 \(g\) 除以 \(h\) 的余式相同.
\end{ex}

\begin{proof}
    充分性. 记 \(f=ph+r,g=qh+r\), 其中 \(\deg r<\deg h\), 则 \(f-g=\p{p-q}h\), 故 \(h\mid f-g\).
    
    必要性. 设 \(f=ph+r_1,g=qh+r_2\), 其中 \(\deg r_1<\deg h,\deg r_2<\deg h\).

    \(f-g=\p{p-q}h+r_1-r_2\). 由 \(h\mid f-g\) 可知 \(h\mid r_1-r_2\),
    又由 \(\deg\p{r_1-r_2}<\deg h\), 可得 \(r_1-r_2=0\), 即 \(r_1=r_2\).
\end{proof}

\begin{ex}{8}\label{ex:1.3.8}
    若 \(x^2+x+1\mid f\p{x^3}+xg\p{x^3}\), 证明 \(x-1\mid f\p{x}\) 且 \(x-1\mid g\p{x}\).
\end{ex}

\begin{proof}
    设 \(f\p{x}=\p{x-1}p\p{x}+r_1\p{x}, g\p{x}=\p{x-1}q\p{x}+r_2\p{x}\), 其中 \(\deg r_1<1,\deg r_2<1\).
    则 \(r_1\p{x},r_2\p{x}\) 为常数.
    
    由余式定理, 有 \(r_1\p{x}=f\p{1},r_2\p{x}=g\p{1}\). 因此
    \begin{equation}
        f\p{x^3}+xg\p{x^3}=\p{x^3-1}\p{p\p{x^3}+xq\p{x^3}}+f\p{1}+xg\p{1}.
        \label{eq:1.3.8-RHS}
    \end{equation}
    由
    \[
    x^3-1=\p{x-1}\p{x^2+x+1},
    \]
    可知
    \[
    x^2+x+1\mid x^3-1.
    \]
    故
    \[
    x^2+x+1\mid\p{x^3-1}\p{p\p{x^3}+xq\p{x^3}}.
    \]
    由题设 \(x^2+x+1\mid f\p{x^3}+xg\p{x^3}\) 及式~\eqref{eq:1.3.8-RHS}, 可得
    \[
    x^2+x+1\mid f\p{1}+xg\p{1}.
    \]
    然而 \(\deg\p{f\p{1}+xg\p{1}}\leq1<2=\deg\p{x^2+x+1}\), 故 \(f\p{1}+xg\p{1}=0\),
    即 \(f\p{1}=g\p{1}=0\), 即 \(1\) 既是 \(f\p{x}\) 的根, 也是 \(g\p{x}\) 的根.
    
    由因式定理, 有
    \[
    x-1\mid f\p{x}, \quad x-1\mid g\p{x}.\qedhere
    \]
\end{proof}
