\section{整除与因式}\label{sec:1.3}

\begin{ex}{1}\label{ex:1.3.1}
    \(f(x)=x^4+x^3+x^2+x+1,g(x)=x^2+2x+1\), 求 \(f(x)\) 除以 \(g(x)\) 的商和余式.
\end{ex}

\begin{solution}
    对 \(f(x)\) 除以 \(g(x)\) 进行多项式长除法:
    \[
    \begin{array}{r @{\;} l @{\;} c @{\;} c @{\;} c @{\;} c @{\;} c @{\;} c @{\;} c @{\;} c @{\;} c}
        & & x^2 & - & x & + & 2 & & & & \\
        \cline{2-11}
        x^2+2x+1 & \big) & x^4 & + & x^3 & + & x^2 & + & x & + & 1 \\
        & & x^4 & + & 2x^3 & + & x^2 & & & & \\
        \cline{3-11}
        & & & - & x^3 & & & + & x & + & 1 \\
        & & & - & x^3 & - & 2x^2 & - & x & & \\
        \cline{4-11}
        & & & & & & 2x^2 & + & 2x & + & 1 \\
        & & & & & & 2x^2 & + & 4x & + & 2 \\
        \cline{7-11}
        & & & & & & & - & 2x & - & 1
    \end{array}
    \]
    因此, 商为 \(x^2-x+2\), 余式为 \(-2x-1\).
\end{solution}

\begin{ex}{3}\label{ex:1.3.3}
    设 \(f,g,h\in\Omega[x]\), 证明 \(h\mid f-g\) 当且仅当 \(f\) 与 \(g\) 除以 \(h\) 的余式相同.
\end{ex}

\begin{proof}
    充分性. 记 \(f=ph+r,g=qh+r\), 其中 \(\deg r<\deg h\), 则 \(f-g=(p-q)h\), 故 \(h\mid f-g\).
    
    必要性. 设 \(f=ph+r_1,g=qh+r_2\), 其中 \(\deg r_1<\deg h,\deg r_2<\deg h\).

    \(f-g=(p-q)h+r_1-r_2\). 由 \(h\mid f-g\) 可知 \(h\mid r_1-r_2\),
    又由 \(\deg(r_1-r_2)<\deg h\), 可得 \(r_1-r_2=0\), 即 \(r_1=r_2\).
\end{proof}

\begin{ex}{8}\label{ex:1.3.8}
    若 \(x^2+x+1\mid f(x^3)+xg(x^3)\), 证明 \(x-1\mid f(x)\) 且 \(x-1\mid g(x)\).
\end{ex}

\begin{proof}
    设 \(f(x)=(x-1)p(x)+r_1(x), g(x)=(x-1)q(x)+r_2(x)\), 其中 \(\deg r_1<1,\deg r_2<1\).
    则 \(r_1(x),r_2(x)\) 为常数.
    
    由余式定理, 有 \(r_1(x)=f(1),r_2(x)=g(1)\). 因此
    \begin{equation}
        f(x^3)+xg(x^3)=(x^3-1)(p(x^3)+xq(x^3))+f(1)+xg(1).
        \label{eq:1.3.8-RHS}
    \end{equation}
    由
    \[
    x^3-1=(x-1)(x^2+x+1),
    \]
    可知
    \[
    x^2+x+1\mid x^3-1.
    \]
    故
    \[
    x^2+x+1\mid(x^3-1)(p(x^3)+xq(x^3)).
    \]
    由题设 \(x^2+x+1\mid f(x^3)+xg(x^3)\) 及式~\eqref{eq:1.3.8-RHS}, 可得
    \[
    x^2+x+1\mid f(1)+xg(1).
    \]
    然而 \(\deg(f(1)+xg(1))\leq1<2=\deg(x^2+x+1)\), 故 \(f(1)+xg(1)=0\),
    即 \(f(1)=g(1)=0\), 即 \(1\) 既是 \(f(x)\) 的根, 也是 \(g(x)\) 的根.
    
    由因式定理, 有
    \[
    x-1\mid f(x), \quad x-1\mid g(x).\qedhere
    \]
\end{proof}
