\section{代数学基本定理}\label{sec:1.7}

\begin{ex}{1}\label{ex:1.7.1}
    设 \(f(x)\in\Omega[x]\) 是 \(n\) 次首一多项式,
    若 \(f(x)\) 的根为 \(2,3,\cdots,n+1\),
    求 \(f(x)\) 的 \(n-1\) 次项系数, 常数项以及 \(f(x)\) 的系数和.
\end{ex}

\begin{solution}
    由题意, 有
    \[
    f(x)=\prod_{i=2}^{n+1}(x-i).
    \]
    设
    \[
    f(x)=\sum_{i=0}^{n}a_{i}x^{n-i},
    \]
    由题意, \(a_0=1\), 由 Vièta 定理, \(f(x)\) 的 \(n-1\) 次项系数
    \[
    c_1=-\sum_{i=2}^{n+1}i=-\frac{n(n+3)}{2},
    \]
    常数项
    \[
    c_n=(-1)^n\prod_{i=2}^{n+1}i=(-1)^n(n+1)!,
    \]
    系数和
    \[
    \sum_{i=0}^{n}a_i=f(1)=\prod_{i=2}^{n+1}(1-i)=(-1)^{n}n!\qedhere
    \]
\end{solution}

\begin{ex}{4}\label{ex:1.7.4}
    设 \(f(x)=x^n+a_1x^{n-1}+\cdots+a_n\) 的 \(n\) 个根为 \(x_1,x_2,\cdots,x_n\).
    求 \(\displaystyle\sum_{i=1}^{n}x_i^{-1}\) 和 \(\displaystyle\sum_{i=1}^{n}x_i^2\).
\end{ex}

\begin{solution}
    由 Vièta 定理, 有
    \begin{gather*}
    \sum_{i=1}^{n}x_i^{-1}
    =-\p{(-1)^{n-1}\sum_{1\leq i_1<i_2<\cdots<i_{n-1}\leq n}\prod_{j=1}^{n-1}x_{i_j}}
    /\p{(-1)^n\prod_{i=1}^{n}x_{i}}=-\frac{a_{n-1}}{a_n},\\
    \sum_{i=1}^{n}x_i^2
    =\p{-\sum_{i=1}^{n}x_i}^2-2\sum_{1\leq i_1<i_2\leq n}x_{i_1}x_{i_2}=a_1^2-2a_2.\qedhere
    \end{gather*}
\end{solution}
